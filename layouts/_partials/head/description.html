{{- /* Returns the page's meta description as plain text, whitespace collapsed and truncated to about 160 characters. Shared by , OpenGraph, Twitter Cards and JSON-LD so all of them say the same thing. */ -}} {{- $raw := .Description | default .Summary | default site.Params.description | default site.Title -}} {{- /* truncate HTML-escapes a plain string; unescape so JSON-LD and llms.txt get plain text (HTML contexts re-escape it). */ -}} {{- $desc := $raw | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace | truncate 160 | htmlUnescape -}} {{- return $desc -}}
